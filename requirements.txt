# Runtime dependencies for the deployed app. Deliberately does NOT include PyTorch:
# the deep model is *scored* in NumPy (lottery_core/deep_runtime.py) from the .npz
# weights in artifacts/, so serving it needs no torch. On Linux the PyPI torch wheel
# pulls ~2.5 GB of bundled CUDA libraries for a CPU-only 160 KB LSTM, which is what
# made cold starts slow. Fitting the model still needs torch -- see requirements-train.txt.
#
# Upper bounds are deliberate: an unpinned resolve is what turns a cold start into a
# multi-minute rebuild and lets a major release land in production unreviewed.
numpy>=1.24,<3.0
pandas>=2.0,<3.0
scikit-learn>=1.3,<2.0
joblib>=1.3,<2.0
streamlit>=1.36,<2.0
plotly>=5.18,<7.0
